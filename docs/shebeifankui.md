     问题 1:WS 代理用明文 TCP 但后端是 HTTPS(阻塞性)                                                                                                                                                                                                                         
     症状                                                                                                                                                                                                                                                                    
     前端 wss://host:6060/ws/asr → 浏览器报 401/连接失败。Node 无日志。直连 8000 正常。                                                                                                                                                                                      
     根因                                                                                                                                                                                                                                                                    
     dist/server/index.js 的 HTTPS server upgrade handler:                                                                                                                                                                                                                   
                                                                                                                                                                                                                                                                             
     js                                                                                                                                                                                                                                                                      
     let s = await import("node:net");          // ← 明文 TCP                                                                                                                                                                                                                
     // ...                                                                                                                                                                                                                                                                  
     let ws = s.connect(8000, "127.0.0.1", cb); // ← 发明文 HTTP                                                                                                                                                                                                             
                                                                                                                                                                                                                                                                             
     但 uvicorn 启动参数带了 --ssl-keyfile --ssl-certfile。8000 收到明文 HTTP 后当作 TLS ClientHello 解析失败 → 断连。                                                                                                                                                       
     修复                                                                                                                                                                                                                                                                    
                                                                                                                                                                                                                                                                             
     diff                                                                                                                                                                                                                                                                    
     - let s = await import("node:net");                                                                                                                                                                                                                                     
     + let s = await import("node:tls");                                                                                                                                                                                                                                     
     - let ws = s.connect(port, host, cb);                                                                                                                                                                                                                                   
     + let ws = s.connect(port, host, { rejectUnauthorized: false }, cb);                                                                                                                                                                                                    
                                                                                                                                                                                                                                                                             
     rejectUnauthorized: false 是因为 8000/8001 用的是 Node 同一份自签证书。                                                                                                                                                                                                 
     验证                                                                                                                                                                                                                                                                    
                                                                                                                                                                                                                                                                             
     python3 -c "                                                                                                                                                                                                                                                            
     import ssl, socket                                                                                                                                                                                                                                                      
     ctx = ssl.create_default_context(); ctx.check_hostname=False; ctx.verify_mode=ssl.CERT_NONE                                                                                                                                                                             
     s = ctx.wrap_socket(socket.socket())                                                                                                                                                                                                                                    
     s.connect(('127.0.0.1', 6060))                                                                                                                                                                                                                                          
     s.send(b'GET /ws/asr HTTP/1.1\r\nHost: 127.0.0.1\r\nUpgrade: websocket\r\nConnection: Upgrade\r\nSec-WebSocket-Key: dGhlIHNhbXBsZSBub25jZQ==\r\nSec-WebSocket-Version: 13\r\n\r\n')                                                                                     
     print(s.recv(1024).split(b'\\r\\n')[0])  # 应输出 b'HTTP/1.1 101 Switching Protocols'                                                                                                                                                                                   
     "                                                                                                                                                                                                                                                                       
                                                                                                                                                                                                                                                                             
     问题 2:start 接口不更新运行中服务的配置                                                                                                                                                                                                                                 
     症状                                                                                                                                                                                                                                                                    
     用户在配置面板改了 dashscopeApiKey → 点录音 → 后台用的还是旧 key。                                                                                                                                                                                                      
     根因                                                                                                                                                                                                                                                                    
     MeetingASRManager.start() 方法:                                                                                                                                                                                                                                         
                                                                                                                                                                                                                                                                             
     ts                                                                                                                                                                                                                                                                      
     async start(config: StartConfig) {                                                                                                                                                                                                                                      
       if (this._isRunning) {                                                                                                                                                                                                                                                
         // 如果没传 OSS 配置,直接 return,不处理 config 变化                                                                                                                                                                                                                 
         return;                                                                                                                                                                                                                                                             
       }                                                                                                                                                                                                                                                                     
       // ... 启动新进程                                                                                                                                                                                                                                                     
     }                                                                                                                                                                                                                                                                       
                                                                                                                                                                                                                                                                             
     但 POST /api/meeting-asr/start 每次都被前端调用(每次点录音),请求体里带完整配置。服务已在跑时,新配置被忽略。                                                                                                                                                             
     修复                                                                                                                                                                                                                                                                    
                                                                                                                                                                                                                                                                             
     diff                                                                                                                                                                                                                                                                    
     async start(config: StartConfig) {                                                                                                                                                                                                                                      
       if (this._isRunning) {                                                                                                                                                                                                                                                
     +   // 把新配置推送给运行中的 Python 进程                                                                                                                                                                                                                               
     +   try { await this.updateConfig(config); } catch (e) { /* log */ }                                                                                                                                                                                                    
         return;                                                                                                                                                                                                                                                             
       }                                                                                                                                                                                                                                                                     
       // ... 启动新进程                                                                                                                                                                                                                                                     
     }                                                                                                                                                                                                                                                                       
                                                                                                                                                                                                                                                                             
     updateConfig() 方法已经存在,它 POST 到 http://127.0.0.1:{port}/api/config,把 dashscopeApiKey、llmApiKey、paraformerWsUrl 等推到 ASR 进程。                                                                                                                              
     问题 3:前端 OSS 配置没有 fallback                                                                                                                                                                                                                                       
     症状                                                                                                                                                                                                                                                                    
     用户改 OSS AccessKeyId/Secret → 不生效,后端用的仍是旧值。                                                                                                                                                                                                               
     根因                                                                                                                                                                                                                                                                    
     $s() 函数构建 start API 请求体时:                                                                                                                                                                                                                                       
                                                                                                                                                                                                                                                                             
     ts                                                                                                                                                                                                                                                                      
     const store = i.asrConfig;                                                                                                                                                                                                                                              
     if (store.ossBucket || store.ossAccessKeyId || store.ossAccessKeySecret) {                                                                                                                                                                                              
       body.ossBucket = store.ossBucket;                                                                                                                                                                                                                                     
       body.ossAccessKeyId = store.ossAccessKeyId;     // ← 只读 store                                                                                                                                                                                                       
       body.ossAccessKeySecret = store.ossAccessKeySecret;                                                                                                                                                                                                                   
     }                                                                                                                                                                                                                                                                       
                                                                                                                                                                                                                                                                             
     但 dashscopeApiKey 有 fallback:                                                                                                                                                                                                                                         
     ts                                                                                                                                                                                                                                                                      
     body.dashscopeApiKey = i.asrConfig.dashscopeApiKey || localState.value;                                                                                                                                                                                                 
                                                                                                                                                                                                                                                                             
     配置面板修改的是本地 state(ue.value),没有同步到 store。OSS 缺了这个 fallback。                                                                                                                                                                                          
     修复                                                                                                                                                                                                                                                                    
                                                                                                                                                                                                                                                                             
     diff                                                                                                                                                                                                                                                                    
     if (store.ossBucket || store.ossAccessKeyId || store.ossAccessKeySecret) {                                                                                                                                                                                              
     - body.ossBucket = store.ossBucket;                                                                                                                                                                                                                                     
     + body.ossBucket = store.ossBucket || localOssBucket.value;                                                                                                                                                                                                             
     - body.ossAccessKeyId = store.ossAccessKeyId;                                                                                                                                                                                                                           
     + body.ossAccessKeyId = store.ossAccessKeyId || localOssAccessKeyId.value;                                                                                                                                                                                              
     - body.ossAccessKeySecret = store.ossAccessKeySecret;                                                                                                                                                                                                                   
     + body.ossAccessKeySecret = store.ossAccessKeySecret || localOssAccessKeySecret.value;                                                                                                                                                                                  
     }

     问题 5:缺少运行中健康保活                                                                                                                                                                                                                                               
     症状                                                                                                                                                                                                                                                                    
     Python 进程(uvicorn)被 OOM kill 后,Node 无感知,前端 WS 断连无错误。                                                                                                                                                                                                     
     修复建议                                                                                                                                                                                                                                                                
     在 MeetingASRManager 中加定时心跳:                                                                                                                                                                                                                                      
                                                                                                                                                                                                                                                                             
     ts                                                                                                                                                                                                                                                                      
     start(config) {                                                                                                                                                                                                                                                         
       // ... 现有启动逻辑                                                                                                                                                                                                                                                   
       this._healthTimer = setInterval(async () => {                                                                                                                                                                                                                         
         try {                                                                                                                                                                                                                                                               
           const resp = await fetch(http://127.0.0.1:${this._asrPort}/healthz);                                                                                                                                                                                              
           if (!resp.ok) throw new Error(healthz ${resp.status});                                                                                                                                                                                                            
         } catch {                                                                                                                                                                                                                                                           
           this._healthFailures++;                                                                                                                                                                                                                                           
           if (this._healthFailures >= 3) {                                                                                                                                                                                                                                  
             // 尝试重启,失败则 emit error                                                                                                                                                                                                                                   
             await this.restart();                                                                                                                                                                                                                                           
           }                                                                                                                                                                                                                                                                 
         }                                                                                                                                                                                                                                                                   
       }, 30_000);                                                                                                                                                                                                                                                           
     }                                                                                                                                                                                                                                                                       
                                                                                                                                                                                                                                                                             
     问题 6:pip install 在 venv 失效时阻塞启动                                                                                                                                                                                                                               
     症状                                                                                                                                                                                                                                                                    
     Node 或设备重启后,meeting-asr 首次启动卡 3-10 分钟(pip install on ARM64)                                                                                                                                                                                                
     修复建议                                                                                                                                                                                                                                                                
     在 ensureVirtualEnv() 中加快速校验 + 缓存校验和:                                                                                                                                                                                                                        
                                                                                                                                                                                                                                                                             
     ts                                                                                                                                                                                                                                                                      
     async ensureVirtualEnv(): string {                                                                                                                                                                                                                                      
       const venvPython = path.join(venv, "bin", "python");                                                                                                                                                                                                                  
       const markerFile = path.join(venv, ".hermes-ready");                                                                                                                                                                                                                  
                                                                                                                                                                                                                                                                             
       // 快速路径:检查 marker 文件 + Python 可执行                                                                                                                                                                                                                          
       if (existsSync(markerFile) && existsSync(venvPython)) {                                                                                                                                                                                                               
         return venvPython;                                                                                                                                                                                                                                                  
       }                                                                                                                                                                                                                                                                     
                                                                                                                                                                                                                                                                             
       // 慢速路径:创建 venv + pip install                                                                                                                                                                                                                                   
       // ...                                                                                                                                                                                                                                                                
       // 安装完成后写 marker                                                                                                                                                                                                                                                
       writeFileSync(markerFile, new Date().toISOString());                                                                                                                                                                                                                  
     }                                                                                                                                                                                                                                                                       
                                                                                                                                                                                                                                                                             
     同时在 /api/meeting-asr/status 里暴露启动阶段:                                                                                                                                                                                                                          
                                                                                                                                                                                                                                                                             
     ts                                                                                                                                                                                                                                                                      
     get status() {                                                                                                                                                                                                                                                          
       return {                                                                                                                                                                                                                                                              
         isRunning: this._isRunning,                                                                                                                                                                                                                                         
         startupPhase: this._startupPhase, // "idle" | "venv" | "pip_install" | "starting" | "ready" | "error"                                                                                                                                                               
         // ...                                                                                                                                                                                                                                                              
       };                                                                                                                                                                                                                                                                    
     }                                                                                                                                                                                                                                                                       
                                                                                                                                                                                                                                                                             
     前端据此显示"正在安装依赖(约 3 分钟)..."而不是"连接中"。
